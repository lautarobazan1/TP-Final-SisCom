#include <RadioLib.h>
#include <Adafruit_GFX.h>
#include <Adafruit_ST7735.h>
#include <SPI.h>

// Definición de pines para la pantalla TFT (ST7735) en la Heltec Wireless Tracker
#define TFT_CS     38
#define TFT_RST    39
#define TFT_DC     40
#define TFT_MOSI   42
#define TFT_SCLK   41
#define TFT_LED    21
#define TFT_CTRL   46
#define VEXT_PIN    3

// Definición de pines para el chip de radio SX1262
#define LORA_NSS   8
#define LORA_DIO1  14
#define LORA_RST   12
#define LORA_BUSY  13

// Instancias de la pantalla y de la Radio
Adafruit_ST7735 tft = Adafruit_ST7735(TFT_CS, TFT_DC, TFT_MOSI, TFT_SCLK, TFT_RST);
SX1262 radio = new Module(LORA_NSS, LORA_DIO1, LORA_RST, LORA_BUSY);

int packetCount = 0;

void setup() {
  Serial.begin(115200);

  // 1. Encender alimentación interna de la pantalla y retroiluminación
  pinMode(VEXT_PIN, OUTPUT);
  digitalWrite(VEXT_PIN, HIGH); // Alimenta periféricos
  pinMode(TFT_CTRL, OUTPUT);
  digitalWrite(TFT_CTRL, LOW);  // Habilita controlador TFT
  pinMode(TFT_LED, OUTPUT);
  digitalWrite(TFT_LED, HIGH); // Encender luz de fondo

  // 2. Inicializar la pantalla TFT (Sin el guion bajo antes de 160)
  tft.initR(INITR_MINI160x80); // Resolución del display de la Tracker (80x160)
  tft.setRotation(1);           // Modo horizontal
  tft.invertDisplay(true);      // Corregir colores invertidos en la Tracker
  tft.fillScreen(ST77XX_BLACK);

  // Dibujar encabezado en pantalla
  tft.setTextColor(ST77XX_YELLOW);
  tft.setTextSize(1);
  tft.setCursor(5, 5);
  tft.println("--- HELTEC LORA ---");
  tft.drawFastHLine(0, 15, 160, ST77XX_WHITE);

  tft.setTextColor(ST77XX_WHITE);
  tft.setCursor(5, 25);
  tft.println("Iniciando radio...");

  // 3. Inicializar el módulo LoRa SX1262 (Mismos parámetros que la Radio Node)
  int state = radio.begin(915.0, 125.0, 7, 5, 0x12, 14);

  tft.setCursor(5, 40);
  if (state == RADIOLIB_ERR_NONE) {
    tft.setTextColor(ST77XX_GREEN);
    tft.println("LoRa OK. Escuchando...");
  } else {
    tft.setTextColor(ST77XX_RED);
    tft.print("Error LoRa: ");
    tft.println(state);
    while (true);
  }
}

void loop() {
  String mensaje = "";

  // Intentar recibir paquete (bloqueante hasta que llega uno)
  int state = radio.receive(mensaje);

  if (state == RADIOLIB_ERR_NONE) {
    packetCount++;

    // Imprimir en Monitor Serie para depurar
    Serial.print("Paquete #");
    Serial.print(packetCount);
    Serial.print(" | Mensaje: ");
    Serial.print(mensaje);
    Serial.print(" | RSSI: ");
    Serial.print(radio.getRSSI());
    Serial.println(" dBm");

    // Limpiar el área central de la pantalla para refrescar el texto
    tft.fillRect(0, 25, 160, 55, ST77XX_BLACK);

    // Mostrar contador y datos
    tft.setCursor(5, 25);
    tft.setTextColor(ST77XX_CYAN);
    tft.print("Pkt #");
    tft.println(packetCount);

    tft.setCursor(5, 40);
    tft.setTextColor(ST77XX_WHITE);
    tft.print("Txt: ");
    tft.println(mensaje);

    // Mostrar nivel de señal (RSSI)
    tft.setCursor(5, 58);
    tft.setTextColor(ST77XX_GREEN);
    tft.print("RSSI: ");
    tft.print(radio.getRSSI());
    tft.println(" dBm");
  }
}
